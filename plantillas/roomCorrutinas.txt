2️⃣ DAO (XXDao.kt)

Ahora todo lo que accede a BD será suspend.

package dao

import androidx.lifecycle.LiveData
import androidx.room.*
import model.XX

@Dao
interface XXDao {

    @Insert
    suspend fun insertar(xx: XX): Long
    // devuelve el ID generado

    @Update
    suspend fun actualizar(xx: XX)

    @Delete
    suspend fun eliminar(xx: XX)

    @Query("SELECT * FROM xx")
    fun getAll(): LiveData<List<XX>>
    // LiveData no necesita suspend

    @Query("SELECT * FROM xx WHERE id = :id")
    suspend fun getById(id: Long): XX?
}
¿Por qué getAll() no lleva suspend?

Porque devuelve LiveData.

LiveData ya observa automáticamente la base de datos en segundo plano.

O sea:

fun getAll(): LiveData<List<XX>>

está bien así.

3️⃣ Base de datos (AppDataBase.kt)

Aquí quitamos allowMainThreadQueries() porque ya usamos corrutinas.

package database

import android.content.Context
import androidx.room.Database
import androidx.room.Room
import androidx.room.RoomDatabase
import dao.XXDao
import model.XX

@Database(
    entities = [XX::class],
    version = 1,
    exportSchema = false
)
abstract class AppDataBase : RoomDatabase() {

    abstract fun xxDao(): XXDao

    companion object {

        @Volatile
        private var INSTANCE: AppDataBase? = null

        fun getDatabase(context: Context): AppDataBase {

            return INSTANCE ?: synchronized(this) {

                val instance = Room.databaseBuilder(
                    context.applicationContext,
                    AppDataBase::class.java,
                    "xx_db"
                ).build()

                INSTANCE = instance
                instance
            }
        }
    }
}
4️⃣ Repository (Repository.kt)

El repository ahora también usa suspend.

package repository

import androidx.lifecycle.LiveData
import dao.XXDao
import model.XX

class Repository(
    private val dao: XXDao
) {

    suspend fun insertar(xx: XX): Long {
        return dao.insertar(xx)
    }

    suspend fun actualizar(xx: XX) {
        dao.actualizar(xx)
    }

    suspend fun eliminar(xx: XX) {
        dao.eliminar(xx)
    }

    fun listaTodo(): LiveData<List<XX>> {
        return dao.getAll()
    }

    suspend fun getById(id: Long): XX? {
        return dao.getById(id)
    }
}
5️⃣ ViewModel (XXViewModel.kt)

Aquí es donde realmente entran las corrutinas.

Usamos:

viewModelScope.launch

para lanzar tareas asincrónicas.

package viewmodel

import android.app.Application
import androidx.lifecycle.*
import database.AppDataBase
import kotlinx.coroutines.launch
import model.XX
import repository.Repository

class XXViewModel(
    application: Application
) : AndroidViewModel(application) {

    private val repository: Repository

    val listaXX: LiveData<List<XX>>

    init {

        val dao =
            AppDataBase
                .getDatabase(application)
                .xxDao()

        repository = Repository(dao)

        listaXX = repository.listaTodo()
    }

    // INSERTAR
    fun insertar(xx: XX) {

        viewModelScope.launch {
            repository.insertar(xx)
        }
    }

    // ACTUALIZAR
    fun actualizar(xx: XX) {

        viewModelScope.launch {
            repository.actualizar(xx)
        }
    }

    // ELIMINAR
    fun eliminar(xx: XX) {

        viewModelScope.launch {
            repository.eliminar(xx)
        }
    }

    // BUSCAR POR ID
    fun getById(
        id: Long,
        callback: (XX?) -> Unit
    ) {

        viewModelScope.launch {

            val resultado =
                repository.getById(id)

            callback(resultado)
        }
    }
}
Cómo usar getById()

Antes hacías:

val plato = viewModel.getById(1)

Pero ahora no se puede porque es asíncrono.

Ahora:

viewModel.getById(1) { plato ->

    if (plato != null) {

        txtNombre.setText(plato.nombre)

    }
}
