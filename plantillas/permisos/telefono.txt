    ActivityResultContracts.RequestPermission()

    if (concedido) {
        telefonoPendiente?.let { llamar(it) }

        Toast.makeText(this, "Permiso denegado", Toast.LENGTH_SHORT).show()

    val intent = Intent(Intent.ACTION_CALL).apply {
