// Senda solo guarda su propia info
@Entity(tableName = "Senda")
data class Senda(
    @PrimaryKey(autoGenerate = true) val id: Long = 0,
    val nombre: String,
    val tefono: String,
    val foto: Int
)

// Actividad tiene la foreign key de Senda
@Entity(tableName = "Actividad")
data class Actividad(
    @PrimaryKey(autoGenerate = true) val id: Long = 0,
    val sendaId: Long,   // ← clave foránea
    val tipo: String,
    val latA: Double,
    val longA: Double
)

// Clase auxiliar para obtenerlas juntas
data class SendaConActividades(
    @Embedded val senda: Senda,
    @Relation(
        parentColumn = "id",
        entityColumn = "sendaId"
    )
    val actividades: List<Actividad>
)
