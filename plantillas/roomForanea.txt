3. RestaurantePlato.kt

@Entity(
    tableName = "RestaurantePlato",
    primaryKeys = ["restauranteId", "platoId"],
    foreignKeys = [
        ForeignKey(
            entity = Restaurante::class,
            parentColumns = ["id"],
            childColumns = ["restauranteId"],
            onDelete = ForeignKey.CASCADE
        ),

        ForeignKey(
            entity = Plato::class,
            parentColumns = ["id"],
            childColumns = ["platoId"],
            onDelete = ForeignKey.CASCADE
        )
    ],
    indices = [Index("platoId")]
)

data class RestaurantePlato(
    val restauranteId: Long,
    val platoId: Long
)

@Dao

    @Query("SELECT * FROM Restaurante WHERE id = :id")
    fun getById(id: Long): Restaurante?

    @Transaction
    @Query("SELECT * FROM Restaurante")
    fun getAllConPlatos(): LiveData<List<RestauranteConPlatos>>

6. AppDataBase.kt

@Database(
    entities = [Restaurante::class, Plato::class, RestaurantePlato::class],
    version = 1,
    exportSchema = false
)
abstract class AppDataBase : RoomDatabase() {

    abstract fun restauranteDao(): RestauranteDao
    abstract fun platoDao(): PlatoDao
    abstract fun restaurantePlatoDao(): RestaurantePlatoDao

    companion object {
        @Volatile private var INSTANCE: AppDataBase? = null

        fun getDatabase(context: Context): AppDataBase {
            return INSTANCE ?: synchronized(this) {
                Room.databaseBuilder(
                    context.applicationContext,
                    AppDataBase::class.java,
                    "restaurante_db"
                )
                .allowMainThreadQueries()
                .build()
                .also { INSTANCE = it }
            }
        }
    }
}

7. RestauranteRepository.kt

class RestauranteRepository(
    private val restauranteDao: RestauranteDao,
    private val platoDao: PlatoDao,
    private val relDao: RestaurantePlatoDao
) {
    fun insertarRestaurante(r: Restaurante): Long = restauranteDao.insertar(r)
    fun insertarPlato(p: Plato): Long = platoDao.insertar(p)

    fun asociarPlato(restauranteId: Long, platoId: Long) {
        relDao.insertar(RestaurantePlato(restauranteId, platoId))
    }

    fun eliminarRestaurante(r: Restaurante) = restauranteDao.eliminar(r)
    fun eliminarPlato(p: Plato) = platoDao.eliminar(p)

    fun listaTodo(): LiveData<List<Restaurante>> = restauranteDao.getAll()
    fun listaTodoConPlatos(): LiveData<List<RestauranteConPlatos>> = restauranteDao.getAllConPlatos()
    fun getRestauranteConPlatos(id: Long): RestauranteConPlatos? = restauranteDao.getConPlatosById(id)
}

8. RestauranteViewModel.kt

class RestauranteViewModel(application: Application) : AndroidViewModel(application) {

    private val repository: RestauranteRepository
    val listaConPlatos: LiveData<List<RestauranteConPlatos>>

    init {
        val db = AppDataBase.getDatabase(application)
        repository = RestauranteRepository(
            db.restauranteDao(),
            db.platoDao(),
            db.restaurantePlatoDao()
        )
        listaConPlatos = repository.listaTodoConPlatos()
    }

    fun insertarRestaurante(r: Restaurante): Long = repository.insertarRestaurante(r)
    fun insertarPlato(p: Plato): Long = repository.insertarPlato(p)
    fun asociarPlato(restauranteId: Long, platoId: Long) = repository.asociarPlato(restauranteId, platoId)
    fun eliminarRestaurante(r: Restaurante) = repository.eliminarRestaurante(r)
    fun getConPlatos(id: Long) = repository.getRestauranteConPlatos(id)
}
