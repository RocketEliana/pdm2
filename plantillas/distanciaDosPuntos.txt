// Dentro de tu ViewModel o Activity
fun calcularDistanciaAsincrona(lat1: Double, lon1: Double, lat2: Double, lon2: Double) {
    // Lanzamos la corrutina en el ámbito del ciclo de vida (lifecycleScope)
    lifecycleScope.launch {
        
        // Cambiamos el contexto a Dispatchers.Default (optimizado para cálculos)
        val distancia = withContext(Dispatchers.Default) {
            haversine(lat1, lon1, lat2, lon2)
        }

        // Al salir del bloque withContext, volvemos al hilo principal automáticamente
        // Ahora podemos actualizar la UI sin riesgo de crash
        tvDistancia.text = "Distancia: ${String.format("%.2f", distancia)} km"
    }
}

// Función de apoyo (la lógica matemática)
fun haversine(lat1: Double, lon1: Double, lat2: Double, lon2: Double): Double {
    val r = 6371.0 // Radio de la Tierra en km
    val dLat = Math.toRadians(lat2 - lat1)
    val dLon = Math.toRadians(lon2 - lon1)
    val a = Math.sin(dLat / 2).pow(2) +
            Math.cos(Math.toRadians(lat1)) * Math.cos(Math.toRadians(lat2)) *
            Math.sin(dLon / 2).pow(2)
    val c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
    return r * c
}
