 private fun mostrarCalendario() {
        val calendario = Calendar.getInstance()
        DatePickerDialog(requireContext(), { _, año, mes, dia ->
            TimePickerDialog(requireContext(), { _, hora, min ->
                val fechaHora = String.format("%02d/%02d/%d %02d:%02d", dia, mes + 1, año, hora, min)
                binding.fechaadd.setText(fechaHora)
            }, calendario.get(Calendar.HOUR_OF_DAY), calendario.get(Calendar.MINUTE), true).show()
        }, calendario.get(Calendar.YEAR), calendario.get(Calendar.MONTH), calendario.get(Calendar.DAY_OF_MONTH)).show()
    }
